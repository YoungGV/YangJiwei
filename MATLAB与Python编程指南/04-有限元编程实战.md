# 第四章 有限元编程实战

## 4.1 MATLAB 桁架有限元程序

完整的平面桁架有限元分析程序：

```matlab
%% 平面桁架有限元分析
% 输入数据
E = 200e3;   % 弹性模量 MPa
A = 500;     % 截面积 mm²

% 节点坐标 [x, y]（mm）
nodes = [0, 0;       % 节点1
         4000, 0;    % 节点2
         2000, 3000]; % 节点3

% 单元连接 [节点i, 节点j]
elements = [1, 2;    % 单元1
            1, 3;    % 单元2
            2, 3];   % 单元3

nNodes = size(nodes, 1);      % 节点数 = 3
nElements = size(elements, 1); % 单元数 = 3
nDOF = 2 * nNodes;             % 总自由度 = 6

%% 组装总刚度矩阵
K = zeros(nDOF, nDOF);

for e = 1:nElements
    n1 = elements(e, 1);
    n2 = elements(e, 2);
    
    x1 = nodes(n1, 1); y1 = nodes(n1, 2);
    x2 = nodes(n2, 1); y2 = nodes(n2, 2);
    
    L = sqrt((x2-x1)^2 + (y2-y1)^2);  % 杆长
    l = (x2-x1)/L;  % cosθ
    m = (y2-y1)/L;  % sinθ
    
    % 单元刚度矩阵（全局坐标，4×4）
    Ke = (E*A/L) * [ l^2,  l*m, -l^2, -l*m;
                     l*m,  m^2, -l*m, -m^2;
                    -l^2, -l*m,  l^2,  l*m;
                    -l*m, -m^2,  l*m,  m^2];
    
    % 自由度编号
    dof = [2*n1-1, 2*n1, 2*n2-1, 2*n2];
    
    % 组装
    K(dof, dof) = K(dof, dof) + Ke;
end

%% 施加载荷
F = zeros(nDOF, 1);
F(6) = -10000;   % 节点3 y方向 -10 kN

%% 施加边界条件
% 节点1: u1=v1=0, 节点2: v2=0
fixedDOF = [1, 2, 4];         % 固定自由度
freeDOF = setdiff(1:nDOF, fixedDOF);  % 自由自由度

%% 求解
u = zeros(nDOF, 1);
u(freeDOF) = K(freeDOF, freeDOF) \ F(freeDOF);

%% 计算各杆内力
fprintf('\n=== 结果 ===\n');
fprintf('节点位移:\n');
for i = 1:nNodes
    fprintf('  节点%d: u=%.4f mm, v=%.4f mm\n', ...
        i, u(2*i-1), u(2*i));
end

fprintf('\n各杆内力:\n');
for e = 1:nElements
    n1 = elements(e, 1);
    n2 = elements(e, 2);
    x1 = nodes(n1,1); y1 = nodes(n1,2);
    x2 = nodes(n2,1); y2 = nodes(n2,2);
    L = sqrt((x2-x1)^2 + (y2-y1)^2);
    l = (x2-x1)/L; m = (y2-y1)/L;
    
    ue = [u(2*n1-1); u(2*n1); u(2*n2-1); u(2*n2)];
    
    % 局部位移 → 内力
    FN = E*A/L * [-l -m l m] * ue;
    sigma = FN / A;
    
    fprintf('  杆%d (%d-%d): F=%.2f N, σ=%.2f MPa\n', ...
        e, n1, n2, FN, sigma);
end
```

## 4.2 Python 桁架有限元程序

```python
import numpy as np

# === 输入数据 ===
E = 200e3    # MPa
A = 500      # mm²

nodes = np.array([
    [0, 0],          # 节点0
    [4000, 0],       # 节点1
    [2000, 3000]     # 节点2
])

elements = np.array([
    [0, 1],   # 单元0
    [0, 2],   # 单元1
    [1, 2]    # 单元2
])

n_nodes = len(nodes)
n_elements = len(elements)
n_dof = 2 * n_nodes

# === 组装刚度矩阵 ===
K = np.zeros((n_dof, n_dof))

for e in range(n_elements):
    n1, n2 = elements[e]
    x1, y1 = nodes[n1]
    x2, y2 = nodes[n2]
    
    L = np.sqrt((x2-x1)**2 + (y2-y1)**2)
    l = (x2-x1) / L
    m = (y2-y1) / L
    
    Ke = (E*A/L) * np.array([
        [ l**2,  l*m, -l**2, -l*m],
        [ l*m,  m**2, -l*m, -m**2],
        [-l**2, -l*m,  l**2,  l*m],
        [-l*m, -m**2,  l*m,  m**2]
    ])
    
    dof = [2*n1, 2*n1+1, 2*n2, 2*n2+1]
    for i in range(4):
        for j in range(4):
            K[dof[i], dof[j]] += Ke[i, j]

# === 载荷 ===
F = np.zeros(n_dof)
F[5] = -10000   # 节点2 y方向

# === 边界条件 ===
fixed = [0, 1, 3]   # 节点0: ux,uy=0; 节点1: uy=0
free = [i for i in range(n_dof) if i not in fixed]

# === 求解 ===
u = np.zeros(n_dof)
u[free] = np.linalg.solve(K[np.ix_(free, free)], F[free])

# === 输出 ===
print("=== 节点位移 ===")
for i in range(n_nodes):
    print(f"  节点{i}: u={u[2*i]:.4f} mm, v={u[2*i+1]:.4f} mm")

print("\n=== 各杆内力 ===")
for e in range(n_elements):
    n1, n2 = elements[e]
    x1, y1 = nodes[n1]
    x2, y2 = nodes[n2]
    L = np.sqrt((x2-x1)**2 + (y2-y1)**2)
    l, m = (x2-x1)/L, (y2-y1)/L
    
    ue = np.array([u[2*n1], u[2*n1+1], u[2*n2], u[2*n2+1]])
    FN = E*A/L * np.array([-l, -m, l, m]) @ ue
    print(f"  杆{e} ({n1}-{n2}): F={FN:.2f} N, σ={FN/A:.2f} MPa")
```

## 4.3 经典 99 行拓扑优化代码

这是结构拓扑优化领域最经典的入门代码，由 Sigmund (2001) 发表。

### MATLAB 版（核心逻辑）

```matlab
function top99(nelx, nely, volfrac, penal, rmin)
%% 99行拓扑优化（SIMP方法）
% nelx, nely: x/y方向单元数
% volfrac: 体积分数约束（如 0.5 = 使用50%材料）
% penal: 惩罚因子（通常取 3）
% rmin: 滤波半径

% 示例调用：top99(60, 20, 0.5, 3, 1.5)

%% 初始化
x = repmat(volfrac, nely, nelx);   % 初始密度场（均匀分布）
loop = 0; change = 1;

%% 有限元分析准备
% 单个单元的刚度矩阵（Q4单元，平面应力）
[KE] = lk;   % 8×8 单元刚度矩阵

%% 优化迭代
while change > 0.01
    loop = loop + 1;
    
    % ① 有限元分析
    [U] = FE(nelx, nely, x, penal, KE);
    
    % ② 目标函数 & 灵敏度
    [KE] = lk;
    c = 0;
    dc = zeros(nely, nelx);
    for ely = 1:nely
        for elx = 1:nelx
            n1 = (nely+1)*(elx-1)+ely;
            n2 = (nely+1)*elx+ely;
            Ue = U([2*n1-1;2*n1;2*n2-1;2*n2;2*n2+1;2*n2+2;2*n1+1;2*n1+2],1);
            
            % 柔度（目标函数）
            c = c + x(ely,elx)^penal * Ue' * KE * Ue;
            
            % 灵敏度
            dc(ely,elx) = -penal * x(ely,elx)^(penal-1) * Ue' * KE * Ue;
        end
    end
    
    % ③ 灵敏度滤波
    [dc] = check(nelx, nely, rmin, x, dc);
    
    % ④ 优化准则法（OC）更新设计变量
    [x] = OC(nelx, nely, x, volfrac, dc);
    
    % 收敛判断
    change = max(max(abs(x - xold)));
    
    % 显示结果
    colormap(gray); imagesc(-x); axis equal; axis tight;
    drawnow;
    fprintf('迭代: %d, 柔度: %f, 体积: %f, 变化: %f\n', ...
        loop, c, mean(x(:)), change);
end
end

%% 单元刚度矩阵
function [KE] = lk
E = 1; nu = 0.3;
k = [1/2-nu/6, 1/8+nu/8, -1/4-nu/12, -1/8+3*nu/8, ...
    -1/4+nu/12, -1/8-nu/8, nu/6, 1/8-3*nu/8];
KE = E/(1-nu^2) * ...
    [k(1) k(2) k(3) k(4) k(5) k(6) k(7) k(8);
     k(2) k(1) k(8) k(7) k(6) k(5) k(4) k(3);
     k(3) k(8) k(1) k(6) k(7) k(4) k(5) k(2);
     k(4) k(7) k(6) k(1) k(8) k(3) k(2) k(5);
     k(5) k(6) k(7) k(8) k(1) k(2) k(3) k(4);
     k(6) k(5) k(4) k(3) k(2) k(1) k(8) k(7);
     k(7) k(4) k(5) k(2) k(3) k(8) k(1) k(6);
     k(8) k(3) k(2) k(5) k(4) k(7) k(6) k(1)];
end
```

### SIMP 方法核心思想

```
  ┌────────────────────────────────────────────┐
  │  SIMP（Solid Isotropic Material with       │
  │        Penalization）                      │
  │                                            │
  │  E(x) = x^p · E₀                          │
  │                                            │
  │  x ∈ [0, 1]：设计变量（密度）              │
  │  x = 0：空洞（去掉材料）                   │
  │  x = 1：实体（保留材料）                   │
  │  p：惩罚因子（通常 p=3）                   │
  │      使中间密度的惩罚更重                  │
  │      促使结果趋向 0-1 分布                 │
  │                                            │
  │  目标函数：最小化柔度 c = F^T · u          │
  │  约束条件：Σ x_e · v_e ≤ V_frac · V_total │
  └────────────────────────────────────────────┘
```

## 4.4 Abaqus Python 脚本

Abaqus 使用 Python 作为脚本语言，可以自动化建模。

```python
# Abaqus Python 脚本示例：创建简支梁并分析
from abaqus import *
from abaqusConstants import *

# 创建模型
myModel = mdb.Model(name='SimpleBeam')

# 创建零件（矩形截面梁）
mySketch = myModel.ConstrainedSketch(name='BeamSketch', sheetSize=200)
mySketch.rectangle(point1=(0, 0), point2=(100, 10))
myPart = myModel.Part(name='Beam', dimensionality=TWO_D_PLANAR,
                      type=DEFORMABLE_BODY)
myPart.BaseShell(sketch=mySketch)

# 定义材料
myMaterial = myModel.Material(name='Steel')
myMaterial.Elastic(table=((200e3, 0.3),))

# 创建截面并赋予
myModel.HomogeneousSolidSection(name='BeamSection',
                                material='Steel', thickness=1.0)
region = myPart.Set(faces=myPart.faces, name='AllFaces')
myPart.SectionAssignment(region=region, sectionName='BeamSection')

# ... 后续：Assembly, Step, Load, Mesh, Job
```

## 5.1 MATLAB 力学可视化

### 绘制变形图

```matlab
% 桁架变形图
nodes_orig = [0 0; 4000 0; 2000 3000];  % 原始节点
u = [0; 0; 0.5; 0; 0.3; -1.2];          % 节点位移

scale = 500;  % 放大倍数
nodes_def = nodes_orig + scale * reshape(u, 2, [])';

elements = [1 2; 1 3; 2 3];

figure; hold on;
for e = 1:size(elements, 1)
    n = elements(e, :);
    % 原始（虚线）
    plot(nodes_orig(n, 1), nodes_orig(n, 2), 'b--', 'LineWidth', 1);
    % 变形后（实线）
    plot(nodes_def(n, 1), nodes_def(n, 2), 'r-', 'LineWidth', 2);
end
plot(nodes_orig(:,1), nodes_orig(:,2), 'bo', 'MarkerSize', 10);
plot(nodes_def(:,1), nodes_def(:,2), 'rs', 'MarkerSize', 10);
legend('原始', '变形后');
axis equal; grid on;
title(sprintf('桁架变形图（放大%d倍）', scale));
```

### 绘制弯矩图

```matlab
% 简支梁弯矩图
L = 5;           % m
F = 10;          % kN（跨中集中力）
x = linspace(0, L, 200);

M = zeros(size(x));
for i = 1:length(x)
    if x(i) <= L/2
        M(i) = F/2 * x(i);
    else
        M(i) = F/2 * (L - x(i));
    end
end

figure;
fill([x, L, 0], [M, 0, 0], [0.85 0.92 1], ...
     'EdgeColor', [0 0.4 0.8], 'LineWidth', 2);
xlabel('x (m)'); ylabel('M (kN·m)');
title('弯矩图');
set(gca, 'YDir', 'reverse');  % 弯矩图惯例（正值向下）
grid on;
```

### 云图（等值线）

```matlab
% 有限元应力云图
[X, Y] = meshgrid(0:0.5:100, 0:0.5:50);
sigma = 200 * sin(pi*X/100) .* sin(pi*Y/50);

figure;
contourf(X, Y, sigma, 20, 'LineColor', 'none');
colorbar; colormap('jet');
xlabel('x (mm)'); ylabel('y (mm)');
title('Von Mises 应力云图 (MPa)');
axis equal;
```

## 5.2 Python 力学可视化

### 桁架变形图

```python
import numpy as np
import matplotlib.pyplot as plt

nodes = np.array([[0, 0], [4000, 0], [2000, 3000]])
u = np.array([0, 0, 0.5, 0, 0.3, -1.2])

scale = 500
nodes_def = nodes + scale * u.reshape(-1, 2)

elements = [[0, 1], [0, 2], [1, 2]]

fig, ax = plt.subplots(figsize=(10, 7))
for e in elements:
    ax.plot(nodes[e, 0], nodes[e, 1], 'b--', linewidth=1, alpha=0.5)
    ax.plot(nodes_def[e, 0], nodes_def[e, 1], 'r-', linewidth=2)

ax.plot(nodes[:, 0], nodes[:, 1], 'bo', markersize=10, label='原始')
ax.plot(nodes_def[:, 0], nodes_def[:, 1], 'rs', markersize=10, label='变形后')
ax.legend(fontsize=12)
ax.set_xlabel('x (mm)')
ax.set_ylabel('y (mm)')
ax.set_title(f'桁架变形图（放大{scale}倍）')
ax.set_aspect('equal')
ax.grid(True)
plt.show()
```

### 3D 曲面图

```python
import numpy as np
import matplotlib.pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

x = np.linspace(0, 10, 50)
y = np.linspace(0, 5, 25)
X, Y = np.meshgrid(x, y)
Z = np.sin(np.pi*X/10) * np.sin(np.pi*Y/5) * 0.5

fig = plt.figure(figsize=(12, 7))
ax = fig.add_subplot(111, projection='3d')
surf = ax.plot_surface(X, Y, Z, cmap='coolwarm', alpha=0.9)

ax.set_ylabel('y (mm)')
ax.set_zlabel('w (mm)')
ax.set_title('板的挠度分布')
plt.show()
```

## 5.4 论文级图表制作技巧

```python
import matplotlib.pyplot as plt
import matplotlib as mpl

# 设置全局字体和风格
plt.rcParams.update({
    'font.size': 14,
    'font.family': 'serif',
    'axes.linewidth': 1.2,
    'lines.linewidth': 2,
    'figure.dpi': 150,
    'savefig.dpi': 300,
    'savefig.bbox': 'tight'
})

# 示例：应力收敛性曲线
mesh_sizes = [100, 500, 1000, 2000, 5000, 10000]
stress_max = [185, 210, 225, 232, 235, 236]
analytical = 236.5

fig, ax = plt.subplots(figsize=(8, 5))
ax.semilogx(mesh_sizes, stress_max, 'bo-', markersize=8, label='FEM 结果')
ax.axhline(y=analytical, color='r', linestyle='--', linewidth=1.5,
           label=f'理论解 ({analytical} MPa)')

ax.set_xlabel('单元数量')
ax.set_ylabel('最大应力 (MPa)')
ax.set_title('网格收敛性分析')
ax.legend()
ax.grid(True, alpha=0.3)

plt.savefig('convergence.png', dpi=300)
plt.show()
```

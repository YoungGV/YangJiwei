## 3.1 线性方程组求解（有限元核心）

### MATLAB 实现

```matlab
% 求解 K*u = F
K = [200 -100 0;
    -100  200 -100;
       0 -100  100];
F = [0; 0; 1000];

% 直接法
u = K \ F;

% 对于大型稀疏矩阵
K_sparse = sparse(K);
u = K_sparse \ F;
```

### Python 实现

```python
import numpy as np
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve

K = np.array([[200, -100, 0],
              [-100, 200, -100],
              [0, -100, 100]])
F = np.array([0, 0, 1000])

u = np.linalg.solve(K, F)

# 大型稀疏矩阵
K_sp = csr_matrix(K)
u = spsolve(K_sp, F)
```

## 3.2 悬臂梁挠度计算

### MATLAB 版

```matlab
% 悬臂梁端部集中力——理论解 vs 数值解
F = 1000;       % N
L = 500;        % mm
E = 200e3;      % MPa
b = 20;         % mm（宽）
h = 40;         % mm（高）
I = b*h^3/12;   % 惯性矩

x = linspace(0, L, 100);

% 理论解
w_theory = F/(6*E*I) * (3*L*x.^2 - x.^3);

% 数值解（欧拉-伯努利梁的差分法）
n = 20;              % 节点数
dx = L / (n-1);
xi = linspace(0, L, n);

% 四阶差分格式：EI * d4w/dx4 = q(x)
% 端部集中力等效为分布力
% 这里用有限差分法

% 绘图对比
figure;
plot(x, w_theory, 'b-', 'LineWidth', 2);
xlabel('x (mm)');
ylabel('挠度 w (mm)');
title('悬臂梁挠度');
grid on;

w_max = F*L^3/(3*E*I);
fprintf('最大挠度 = %.4f mm\n', w_max);
```

### Python 版

```python
import numpy as np
import matplotlib.pyplot as plt

F = 1000        # N
L = 500         # mm
E = 200e3       # MPa
b, h = 20, 40   # mm
I = b * h**3 / 12

x = np.linspace(0, L, 100)
w = F / (6*E*I) * (3*L*x**2 - x**3)

plt.figure(figsize=(10, 5))
plt.plot(x, w, 'b-', linewidth=2)
plt.xlabel('x (mm)', fontsize=13)
plt.ylabel('挠度 w (mm)', fontsize=13)
plt.title('悬臂梁挠度曲线', fontsize=15)
plt.grid(True)
plt.show()

print(f"最大挠度 = {F*L**3/(3*E*I):.4f} mm")
```

## 3.3 振动问题——固有频率计算

### 弹簧-质量系统

```
    k₁      k₂      k₃
  ─/\/\/─●─/\/\/─●─/\/\/─▌
          m₁       m₂
```

```matlab
%% MATLAB：求固有频率
m1 = 1; m2 = 1;        % kg
k1 = 100; k2 = 200; k3 = 100;  % N/m

M = [m1 0; 0 m2];       % 质量矩阵
K = [k1+k2, -k2; -k2, k2+k3];  % 刚度矩阵

% 广义特征值问题：K*phi = omega^2 * M * phi
[Phi, Lambda] = eig(K, M);
omega = sqrt(diag(Lambda));  % 固有角频率
f = omega / (2*pi);          % 固有频率 (Hz)

fprintf('第1阶固有频率: %.2f Hz\n', f(1));
fprintf('第2阶固有频率: %.2f Hz\n', f(2));
```

```python
## Python 版
import numpy as np

m1, m2 = 1, 1
k1, k2, k3 = 100, 200, 100

M = np.array([[m1, 0], [0, m2]])
K = np.array([[k1+k2, -k2], [-k2, k2+k3]])

# 广义特征值问题
from scipy.linalg import eigh
eigenvalues, eigenvectors = eigh(K, M)
omega = np.sqrt(eigenvalues)
freq = omega / (2 * np.pi)

for i, f in enumerate(freq):
    print(f"第{i+1}阶固有频率: {f:.2f} Hz")
```

## 3.4 微分方程数值解

### 弹簧-阻尼振动系统

```
  mẍ + cẋ + kx = F(t)
  
  转化为一阶方程组：
  y₁ = x,  y₂ = ẋ
  ẏ₁ = y₂
  ẏ₂ = (F(t) - c·y₂ - k·y₁) / m
```

```python
import numpy as np
from scipy.integrate import solve_ivp
import matplotlib.pyplot as plt

m, c, k = 1.0, 0.5, 100.0   # kg, N·s/m, N/m
F0 = 10.0                    # 阶跃力 N

def vibration(t, y):
    x, v = y
    F = F0 if t >= 0 else 0
    return [v, (F - c*v - k*x) / m]

sol = solve_ivp(vibration, [0, 5], [0, 0],
                t_eval=np.linspace(0, 5, 1000))

plt.figure(figsize=(10, 5))
plt.plot(sol.t, sol.y[0], 'b-', linewidth=2)
plt.xlabel('时间 (s)')
plt.ylabel('位移 x (m)')
plt.title('弹簧-阻尼系统阶跃响应')
plt.grid(True)
plt.axhline(y=F0/k, color='r', linestyle='--', label=f'稳态值 {F0/k:.3f} m')
plt.legend()
plt.show()
```

## 3.5 数值积分在力学中的应用

### 梁的等效节点力

```python
import numpy as np

# 三角形分布载荷 q(x) = q0 * x/L 的等效节点力
# 使用高斯积分
q0 = 10  # kN/m
L = 4    # m

# 2 点高斯积分
xi_gauss = [-1/np.sqrt(3), 1/np.sqrt(3)]
w_gauss = [1, 1]

# 映射到 [0, L]: x = L/2 * (1 + xi)
# 形函数: N1 = (1-xi)/2, N2 = (1+xi)/2

F1 = 0  # 节点1等效力
F2 = 0  # 节点2等效力

for xi, w in zip(xi_gauss, w_gauss):
    x = L/2 * (1 + xi)
    q = q0 * x / L
    N1 = (1 - xi) / 2
    N2 = (1 + xi) / 2
    F1 += w * q * N1 * L/2
    F2 += w * q * N2 * L/2

print(f"节点1等效力: F1 = {F1:.4f} kN")
print(f"节点2等效力: F2 = {F2:.4f} kN")
print(f"总力 = {F1+F2:.4f} kN (理论值 = {q0*L/2:.4f} kN)")
```

## 3.6 非线性方程求解

### 牛顿-拉夫逊法（Newton-Raphson）

```
  有限元中非线性问题的标准求解方法：
  
  K_T(u) · Δu = F - F_int(u)
  u_{n+1} = u_n + Δu
  
  反复迭代直到残差 ||F - F_int|| < 容差
```

```python
import numpy as np

# 示例：求解 f(x) = x^3 - 2x - 5 = 0
def f(x):
    return x**3 - 2*x - 5

def df(x):
    return 3*x**2 - 2

x = 2.0  # 初始猜测
for i in range(20):
    dx = -f(x) / df(x)
    x = x + dx
    print(f"迭代 {i+1}: x = {x:.10f}, f(x) = {f(x):.2e}")
    if abs(f(x)) < 1e-12:
        print(f"收敛！x = {x:.10f}")
        break
```

## 本章公式与方法总结

| 方法 | 用途 | MATLAB | Python |
|------|------|--------|--------|
| 直接解法 | K·u=F | `u=K\F` | `np.linalg.solve(K,F)` |
| 稀疏解法 | 大型方程组 | `sparse` + `\` | `scipy.sparse.linalg.spsolve` |
| 特征值 | 固有频率 | `eig(K,M)` | `scipy.linalg.eigh(K,M)` |
| ODE | 振动/运动 | `ode45` | `scipy.integrate.solve_ivp` |
| 数值积分 | 等效节点力 | `integral` | `scipy.integrate.quad` |
| 非线性 | 牛顿法 | 手动循环 | 手动循环 / `fsolve` |
